# The exact user-1 pause path writes the three SW_RSV0/1/2 backing locations
# with bit 13, mirrors each value into the CSRAM shadow, waits 10 us, then
# polls the three SW_RSV3/4/5 status locations for bit 15. The loop is 200
# iterations at a 10 us delay, i.e. the vendor 2 ms bound. On success it
# disables the DVFSP I2C_APPM clock reference while the caller retains the
# controller's reference.
pause_sw_pause=three_clusters;bit13;CSRAM_mirror
pause_delay_us=10
pause_fw_done=three_status_words;bit15
pause_timeout_us=2000
pause_success=clk_disable_DVFSP_I2C_APPM_reference

# Release clears the pause-source map. Only when the map reaches zero does it
# enable the DVFSP I2C_APPM reference, clear bit 13 on enabled clusters,
# mirror the values, record completion, and wake waiters. Clock-enable failure
# is a fatal path in the retained implementation.
release_map=clear_user1_bit_0x2
release_zero_map=clk_enable;clear_SW_PAUSE_enabled_clusters;CSRAM_mirror;wake_waiters
release_clock_failure=fatal
